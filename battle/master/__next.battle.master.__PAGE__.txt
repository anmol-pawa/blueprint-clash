int openIds = 0;
int errors = 0;
bool live[1024];
}  // namespace
int open(const std::string&) {
  int id = nextId++;
  live[id] = true;
  ++openIds;
  return id;
}
void close(int id) {
  if (id <= 0 || id >= 1024 || !live[id]) {
    ++errors;
    return;
  }
  live[id] = false;
  --openIds;
}
int openCount() { return openIds; }
int closeErrors() { return errors; }
void reset() {
  nextId = 1;
  openIds = 0;
  errors = 0;
  for (bool& b : live) b = false;
}
}  // namespace os
